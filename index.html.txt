<script>

const SHEET_API = "https://opensheet.elk.sh/1Qs3QiinhTlOdH-FT-Jhs6BdYLJ16zegKQK6PMSwF48s/Sheet1";

const params = new URLSearchParams(window.location.search);
const objectID = params.get("id");

fetch(SHEET_API)
.then(res => res.json())
.then(data => {

  const obj = data.find(item => item.ObjectID === objectID);

  if(obj){

    document.getElementById("content").innerHTML = `
      <h1>${obj.ObjectName}</h1>

      <p><b>Object ID:</b> ${obj.ObjectID}</p>
      <p><b>Location:</b> ${obj.Location}</p>
      <p><b>Status:</b> ${obj.Status}</p>
      <p><b>Owner:</b> ${obj.Owner}</p>
    `;

  } else {

    document.getElementById("content").innerHTML =
      "<h1>Object Not Found</h1>";

  }

});

</script>
